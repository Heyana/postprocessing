# 选择性着色器实现：从AOEffectOrigin到AOEffect的演进

本文档通过对比分析AOEffectOrigin和改进后的AOEffect及其着色器代码，详细说明如何实现一个选择性着色器，以环境光遮蔽(AO)效果为例。我们将逐步解析每一个关键改动及其背后的设计原理。

## 1. 基础着色器与选择性着色器对比

首先，让我们对比原始的AO着色器与改进后的选择性AO着色器代码：

### 原始着色器 (ao_compose_origin.frag)

```glsl
uniform sampler2D inputTexture;
uniform highp sampler2D depthTexture;
uniform float power;
uniform vec3 color;

void mainImage(const in vec4 inputColor, const in vec2 uv, out vec4 outputColor) {
  float unpackedDepth = textureLod(depthTexture, uv, 0.).r;

  float ao = unpackedDepth > 0.9999 ? 1.0 : textureLod(inputTexture, uv, 0.0).a;
  ao = pow(ao, power);

  vec3 aoColor = mix(color, vec3(1.), ao);

  aoColor *= inputColor.rgb;

  outputColor = vec4(aoColor, inputColor.a);
}
```

### 改进后的选择性着色器 (ao_compose.frag)

```glsl
uniform sampler2D inputTexture;  // AO的纹理
uniform sampler2D inputBuffer;   // 原始场景的颜色纹理
uniform highp sampler2D depthTexture;
uniform float power;
uniform vec3 color;
uniform float brightnessThreshold; // 亮度阈值，超过此值将减少AO效果
uniform int debugMode; // 调试模式: 0=正常, 1=显示亮度

// 忽略超亮物体的阈值
#define IGNORE_BRIGHTNESS_THRESHOLD 20.0

void mainImage(const in vec4 inputColor, const in vec2 uv, out vec4 outputColor) {
  // 初始默认值，确保在纹理问题时有合理的回退值
  float ao = 1.0;

  // 读取原始场景颜色（用于调试模式）
  vec4 sceneColor = texture2D(inputBuffer, uv);
  
  // 计算场景颜色的亮度用于调试
  float sceneBrightness = max(max(sceneColor.r, sceneColor.g), sceneColor.b);
  
  // 计算输入颜色的亮度
  float brightness = max(max(inputColor.r, inputColor.g), inputColor.b);
  
  // 检测超亮物体（被标记为不参与AO的对象）
  if (brightness > IGNORE_BRIGHTNESS_THRESHOLD) {
    // 如果检测到超亮物体，直接返回原始颜色，不应用AO
    outputColor = inputColor;
    return;
  }

  // AO效果处理
  // 检查uv坐标是否在有效范围内
  if(uv.x >= 0.0 && uv.x <= 1.0 && uv.y >= 0.0 && uv.y <= 1.0) {
    // 读取深度值
    float unpackedDepth = textureLod(depthTexture, uv, 0.).r;
    
    // 检查深度有效性
    if(unpackedDepth > 0.0 && unpackedDepth <= 1.0) {
      if(unpackedDepth > 0.9999) {
        // 深度接近远平面，认为是天空或远处物体，使用1.0（无AO）
        ao = 1.0;
      } else {
        // 读取AO值并检查有效性
        vec4 aoSample = textureLod(inputTexture, uv, 0.0);
        if(aoSample.a >= 0.0 && aoSample.a <= 1.0) {
          ao = aoSample.a;
          ao = pow(ao, power);
        }
      }
    }
  }

  // 计算输入颜色的亮度 (使用感知亮度公式)
  // 在调试模式下使用场景颜色，正常模式下使用输入颜色
  vec3 colorForBrightness = sceneColor.rgb; // 始终使用sceneColor计算亮度
  float perceptualBrightness = dot(colorForBrightness, vec3(0.299, 0.587, 0.114));
  
  // 根据亮度调整AO效果强度，亮度越高AO影响越小
  float aoStrength = clamp(1.0 - (perceptualBrightness / max(brightnessThreshold, 0.001)), 0.0, 1.0);
  
  // 在高亮处减弱AO效果
  ao = mix(1.0, ao, aoStrength);

  // 亮度调试模式
  if (debugMode == 1) {
    // 显示亮度值
    float normalizedBrightness = clamp(perceptualBrightness / brightnessThreshold, 0.0, 1.0);
    outputColor = vec4(vec3(normalizedBrightness), 1.0);
    return;
  } else if (debugMode == 2) {
    // 显示AO强度值
    outputColor = vec4(vec3(aoStrength), 1.0);
    return;
  } else if (debugMode == 3) {
    // 显示最终AO值
    outputColor = vec4(vec3(ao), 1.0);
    return;
  }

  // 正常模式: 应用AO效果
  vec3 aoColor = mix(color, vec3(1.0), ao);
  
  // 混合策略: 使用inputColor作为基础颜色，但应用根据sceneColor计算的AO强度
  aoColor *= inputColor.rgb;

  // 输出最终颜色，确保alpha值正确
  outputColor = vec4(aoColor, inputColor.a);
}
```

### 改进后的 AOEffect 类的关键部分

```javascript
export class AOEffect extends Effect {
    constructor(composer, camera, scene, aoPass, options = defaultAOOptions) {
        // 合并选项
        options = {
            ...defaultAOOptions,
            ...options
        };

        // 使用AO着色器作为基础效果着色器
        super("AOEffect", ao_compose, {
            type: "FinalAOMaterial",
            uniforms: new Map([
                ["inputTexture", new Uniform(null)],
                ["inputBuffer", new Uniform(null)],  // 新增：原始场景纹理
                ["depthTexture", new Uniform(null)],
                ["power", new Uniform(0)],
                ["color", new Uniform(new Color("black"))],
                ["brightnessThreshold", new Uniform(0.7)],  // 新增：亮度阈值
                ["debugMode", new Uniform(0)]  // 新增：调试模式
            ])
        });

        // 存储上次大小设置
        this.lastSize = {
            width: 0,
            height: 0,
            resolutionScale: 0
        };

        // 保存引用
        this.composer = composer;
        this.aoPass = aoPass;
        this.options = options;
        this.scene = scene;
        this.camera = camera;

        // 初始化Selection对象 - 新增：用于选择排除AO的对象
        this._ignoreSelection = options.ignoreSelection || new Selection();

        // 存储原始材质状态的映射 - 新增：用于暂存和恢复材质
        this.originalMaterials = new Map();

        // 创建专用于AO的渲染目标 - 新增：独立渲染目标
        this.createRenderTargets();

        // 创建渲染通道 - 新增：单独的渲染通道
        this.renderPass = new RenderPass(scene, camera);
        this.renderPass.clear = true;

        // 清除通道，用于在不同渲染步骤间清除
        this.clearPass = new ClearPass(true, false, false);

        // 存储原始发光值的映射
        this.originalEmissives = new Map();

        // 存储对象的原始图层设置
        this.originalLayers = new Map();

        // 设置深度纹理
        if (!composer.depthTexture) composer.createDepthTexture();
        this.aoPass.fullscreenMaterial.uniforms.depthTexture.value = composer.depthTexture;
        this.uniforms.get("depthTexture").value = composer.depthTexture;
        
        // ...更多初始化代码...
    }

    // 准备场景，设置对象图层
    prepareScene() {
        // 存储原始材质和发光值
        this.originalMaterials = new Map();

        // 获取要忽略的对象列表
        const selectionItems = this.getSelectionItems();

        // 为忽略列表中的对象应用特殊材质或标记
        selectionItems.forEach(object => {
            if (object && object.isMesh) {
                // 保存原始材质
                this.originalMaterials.set(object.uuid, {
                    material: object.material,
                    emissive: object.material.emissive ? object.material.emissive.clone() : null
                });

                // 设置超高亮度，但保持原始材质外观
                if (object.material.emissive) {
                    object.material._originalEmissive = object.material.emissive.clone();
                    // 使用超高值确保在着色器中被识别
                    object.material.emissive.set(2000, 2000, 2000);
                }
            }
        });
    }

    // 恢复场景原始状态
    restoreScene() {
        // 恢复原始材质
        if (this.originalMaterials) {
            for (const [uuid, data] of this.originalMaterials.entries()) {
                const object = this.scene.getObjectByProperty('uuid', uuid);
                if (object) {
                    // 恢复发光值
                    if (object.material._originalEmissive) {
                        object.material.emissive.copy(object.material._originalEmissive);
                        delete object.material._originalEmissive;
                    }
                }
            }
            this.originalMaterials.clear();
        }
    }

    update(renderer, inputBuffer, outputBuffer, deltaTime, stencilTest, depthPass) {
        // 准备场景 - 设置高亮对象等
        this.prepareScene();

        // 执行渲染前回调
        this.options?.renderBefore?.(this.scene);

        // 检查是否使用TRAA动画噪声
        const hasTRAA = this.composer.passes.some(pass => {
            const effects = pass.effects;
            return pass.enabled && !pass.skipRendering &&
                effects && effects.some(effect => effect instanceof TRAAEffect);
        });

        // 设置动画噪声
        if (hasTRAA && !("animatedNoise" in this.aoPass.fullscreenMaterial.defines)) {
            this.aoPass.fullscreenMaterial.defines.animatedNoise = "";
            this.aoPass.fullscreenMaterial.needsUpdate = true;
        } else if (!hasTRAA && "animatedNoise" in this.aoPass.fullscreenMaterial.defines) {
            delete this.aoPass.fullscreenMaterial.defines.animatedNoise;
            this.aoPass.fullscreenMaterial.needsUpdate = true;
        }

        // 强制清理所有渲染目标
        this.clearAllRenderTargets(renderer);

        // 保存当前渲染目标以便恢复
        const currentRenderTarget = renderer.getRenderTarget();

        // 步骤1: 渲染场景到专用深度缓冲区（所有模型，包括高亮标记的）
        this.renderPass.render(renderer, this.renderTargetAO, this.renderTargetAO, deltaTime, true);

        // 重要: 立即恢复场景状态，避免对象变白问题
        this.restoreScene();

        // 从渲染目标中获取深度纹理和颜色纹理
        const filteredDepthTexture = this.renderTargetAO.depthTexture;

        // 设置原始场景纹理 - 必须在所有模式下都设置
        this.uniforms.get("inputBuffer").value = this.renderTargetAO.texture;

        // 步骤2: 使用过滤后的深度缓冲区渲染AO效果
        this.aoPass.fullscreenMaterial.uniforms.depthTexture.value = filteredDepthTexture;

        // 渲染AO效果
        this.aoPass.render(renderer);

        // 对AO结果进行降噪
        this.poissionDenoisePass.render(renderer);

        // 步骤3: 设置最终AO纹理
        this.uniforms.get("inputTexture").value = this.poissionDenoisePass.texture;

        // 恢复原始渲染目标
        renderer.setRenderTarget(currentRenderTarget);
    }
}
```

## 3. 关键改动分析与实现步骤

下面我们将分步骤分析从标准着色器向选择性着色器的演进过程，详细解释每一步的实现原理和目的。

### 步骤1：添加必要的Uniform变量

**改动内容**：
```javascript
// 原始代码
uniforms: new Map([
    ["inputTexture", new Uniform(null)], 
    ["depthTexture", new Uniform(null)], 
    ["power", new Uniform(0)], 
    ["color", new Uniform(new Color("black"))]
])

// 改进代码
uniforms: new Map([
    ["inputTexture", new Uniform(null)],
    ["inputBuffer", new Uniform(null)],  // 新增
    ["depthTexture", new Uniform(null)],
    ["power", new Uniform(0)],
    ["color", new Uniform(new Color("black"))],
    ["brightnessThreshold", new Uniform(0.7)],  // 新增
    ["debugMode", new Uniform(0)]  // 新增
])
```

**实现原理**：
1. **inputBuffer**：添加这个uniform用于存储原始场景的渲染结果，它是实现选择性处理的关键。通过访问原始场景颜色，我们可以检测哪些像素属于被标记的对象。
2. **brightnessThreshold**：这个阈值控制基于亮度的自适应AO强度，允许在明亮区域减弱AO效果。
3. **debugMode**：添加调试模式，便于可视化不同处理阶段的结果，这对于开发和调试复杂着色器至关重要。

**实现目的**：为选择性处理和自适应效果提供必要的输入数据。

### 步骤2：创建专用渲染目标和渲染通道

**改动内容**：
```javascript
// 新增代码
// 创建专用于AO的渲染目标

// 创建渲染通道

// 清除通道，用于在不同渲染步骤间清除
```

**实现原理**：
1. 创建专用的渲染目标(RenderTarget)，用于捕获包含标记对象的场景渲染结果。
2. 设置专用的RenderPass，负责将场景渲染到我们的特殊渲染目标。
3. 添加ClearPass用于在不同渲染阶段之间清除渲染目标，确保没有残留数据。

**实现目的**：
1. 获取对渲染过程的完全控制
2. 能够在标准渲染管线之外执行特殊渲染操作
3. 确保每次渲染都有干净的起点

### 步骤4：场景准备与恢复

**改动内容**：

**实现原理**：
1. `prepareScene`方法临时修改选中对象的材质，将它们的emissive值设置为极高的值(2000)
2. 这些超高亮度对象在渲染时会产生非常亮的颜色，可以在着色器中通过亮度阈值检测到
3. `restoreScene`方法在渲染完成后立即恢复这些对象的原始材质，避免后续渲染受到影响

**实现目的**：
1. 创建一种标记机制，允许着色器识别特定对象
2. 确保这种标记只在我们的特殊渲染中有效，不影响正常渲染

### 步骤8：改进update方法实现多阶段处理

**改动内容**：
```javascript
update(renderer, inputBuffer, outputBuffer, deltaTime, stencilTest, depthPass) {
    // 1. 准备场景
    this.prepareScene();
    
    // 2. 清理渲染目标
    this.clearAllRenderTargets(renderer);
    
    // 3. 渲染场景到专用渲染目标
    this.renderPass.render(renderer, this.renderTargetAO, this.renderTargetAO, deltaTime, true);
    
    // 4. 立即恢复场景状态

    // 5. 设置纹理

    this.uniforms.get("inputTexture").value = this.poissionDenoisePass.texture;
}
```

**实现原理**：
1. 将整个处理流程分为多个明确的阶段
2. 每个阶段都有特定的职责和输出
3. 确保场景状态正确恢复，避免副作用
4. 正确设置各个纹理输入，确保着色器有正确的数据源

**实现目的**：实现一个健壮、高效的多阶段处理流程，确保选择性着色器能够正常工作。

## 4. 混合策略详解

选择性着色器的一个关键设计决策是混合策略。我们的混合策略解决了以下三个问题：

### 问题1: 分辨率问题
如果只使用`sceneColor`(来自renderTargetAO)作为基础，最终效果会受限于专用渲染目标的分辨率，可能导致画面模糊或像素化。

### 问题2: "对象变白"问题
如果在渲染过程中不立即恢复场景状态，被标记的对象会在其他渲染通道中保持高亮状态，导致它们在最终输出中显示为白色。

### 问题3: 亮度过滤失效问题
如果只使用`inputColor`计算亮度，将无法正确应用基于原始场景亮度的自适应AO效果。

### 混合策略解决方案:

```glsl
// 使用sceneColor计算亮度和AO调整
vec3 colorForBrightness = sceneColor.rgb;
float aoStrength = /* 基于sceneColor的亮度计算 */;

// 但最终效果应用到inputColor上，保持高分辨率和前序效果
vec3 aoColor = mix(color, vec3(1.0), ao);
aoColor *= inputColor.rgb;
```

这种混合策略实现了：
1. 保持高分辨率和前序效果（使用inputColor作为基础）
2. 正确应用亮度自适应的AO效果（使用sceneColor计算亮度）
3. 避免对象变白问题（通过及时恢复场景状态）

## 5. 实现选择性着色器的通用步骤

总结一下，实现选择性着色器的通用步骤如下：

1. **设计标记机制**：确定如何标记和识别需要特殊处理的对象（我们使用emissive设置超高值）

2. **扩展Uniform变量**：添加必要的uniform变量，如原始场景纹理、控制参数和调试选项

3. **创建专用渲染管线**：设置独立的渲染目标和渲染通道，获取对渲染过程的完全控制

4. **实现对象处理逻辑**：编写准备和恢复场景的方法，确保正确标记对象并在处理后恢复

5. **设计着色器识别逻辑**：在着色器中添加检测代码，识别标记的对象并应用特殊处理

6. **添加自适应效果**：根据场景特征（如亮度）动态调整效果强度，提高真实感

7. **实现混合策略**：设计智能混合策略，解决分辨率、材质保真度和效果准确性等问题

8. **添加调试功能**：实现多种调试模式，便于可视化不同处理阶段的结果

9. **优化多阶段处理**：设计清晰的多阶段处理流程，确保各个阶段正确执行并输出预期结果

## 数据流处理与传递机制

在PostProcessing框架中，着色器效果之间的数据流转是通过一系列缓冲区和纹理传递完成的。理解这一流程对实现选择性着色器至关重要。

### 输入数据获取

1. **EffectPass的数据传递**：
   每个效果器（如AOEffect）都会在EffectPass的render方法中被调用其update方法。这个方法接收三个关键参数：
   ```javascript
   effect.update(renderer, inputBuffer, deltaTime, depthPass);
   ```
   其中，inputBuffer包含了上一个通道处理后的渲染结果。

2. **特殊纹理的获取**：
   除了基本的颜色信息外，效果器通常还需要获取特殊纹理，如深度纹理：
   ```javascript
   // 从EffectComposer获取深度纹理
   if (!composer.depthTexture) composer.createDepthTexture();
   this.uniforms.get("depthTexture").value = composer.depthTexture;
   ```

### 数据处理流程

选择性着色器的数据处理流程通常包含以下步骤：

1. **场景准备**：
   ```javascript
   // 标记不参与AO计算的对象
   this.prepareScene();
   ```

2. **渲染到专用缓冲区**：
   ```javascript
   // 渲染场景到专用深度缓冲区
   this.renderPass.render(renderer, this.renderTargetAO, this.renderTargetAO, deltaTime, true);
   ```

3. **恢复场景状态**：
   ```javascript
   // 恢复对象的原始材质状态
   this.restoreScene();
   ```

4. **应用主效果处理**：
   ```javascript
   // 使用深度缓冲区渲染AO效果
   this.aoPass.fullscreenMaterial.uniforms.depthTexture.value = filteredDepthTexture;
   this.aoPass.render(renderer);
   ```

5. **后处理**（如降噪）：
   ```javascript
   // 对AO结果进行降噪
   this.poissionDenoisePass.render(renderer);
   ```

### 数据传递机制

效果器通过修改着色器的uniform变量将处理后的数据传递给着色器程序：

1. **设置输入纹理**：
   ```javascript
   // 设置AO纹理
   this.uniforms.get("inputTexture").value = this.poissionDenoisePass.texture;
   
   // 设置原始场景颜色纹理
   this.uniforms.get("inputBuffer").value = this.renderTargetAO.texture;
   ```

2. **着色器内的数据使用**：
   在着色器代码中，通过uniform变量读取这些纹理数据：
   ```glsl
   uniform sampler2D inputTexture;  // AO的纹理
   uniform sampler2D inputBuffer;   // 原始场景的颜色纹理
   
   // 在mainImage函数中使用这些纹理
   vec4 aoSample = textureLod(inputTexture, uv, 0.0);
   vec4 sceneColor = texture2D(inputBuffer, uv);
   ```

### 输出结果传递

选择性着色器的处理结果最终会通过以下方式传递给下一个效果：

1. **直接输出**：在mainImage函数最后，计算结果被写入到outputColor变量：
   ```glsl
   // 计算最终颜色并输出
   vec3 aoColor = mix(color, vec3(1.0), ao);
   aoColor *= inputColor.rgb;
   outputColor = vec4(aoColor, inputColor.a);
   ```

2. **链式传递**：这个输出结果会成为下一个效果器的输入（inputColor）：
   ```glsl
   void mainImage(const in vec4 inputColor, const in vec2 uv, out vec4 outputColor) {
     // inputColor是上一个效果的输出
     // outputColor将作为下一个效果的输入
   }
   ```

### 与原始实现的区别

相比于原始AOEffect的实现（AOEffectOrigin），选择性AOEffect（AOEffect）的主要数据流差异在于：

1. **多一层渲染目标**：
   ```javascript
   // 原始AOEffect直接使用上一通道的输入
   this.uniforms.get("inputTexture").value = this.poissionDenoisePass.texture;
   
   // 选择性AOEffect使用专用渲染目标的结果
   this.uniforms.get("inputBuffer").value = this.renderTargetAO.texture;
   ```

2. **着色器中的额外处理逻辑**：
   ```glsl
   // 原始shader简单混合AO
   vec3 aoColor = mix(color, vec3(1.), ao);
   
   // 选择性shader增加了亮度检测和过滤
   if (brightness > IGNORE_BRIGHTNESS_THRESHOLD) {
     outputColor = inputColor;
     return;
   }
   ```

理解这一数据流程对于开发其他选择性效果（如选择性景深、选择性辉光）至关重要。通过控制数据的流转路径和处理逻辑，可以实现对指定对象应用特定效果，同时保持其他对象不受影响。
